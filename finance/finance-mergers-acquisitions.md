def merger_dcf(free_cash_flows, wacc, terminal_growth, synergy_pct):
    terminal_value = free_cash_flows[-1] * (1 + terminal_growth) / (wacc - terminal_growth)
    pv_fcfs = sum(fcf / (1 + wacc)**(i+1) for i, fcf in enumerate(free_cash_flows))
    return pv_fcfs + terminal_value / (1 + wacc)**len(free_cash_flows)
